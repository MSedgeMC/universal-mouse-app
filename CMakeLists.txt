cmake_minimum_required(VERSION 3.22)
project(UniversalMouseApp VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets DBus Concurrent Svg)
add_library(mouse-core src/core/protocol.cpp src/core/storage.cpp src/core/profile.cpp src/drivers/device.cpp src/platform/linux.cpp)
target_include_directories(mouse-core PUBLIC src)
target_link_libraries(mouse-core PUBLIC Qt6::Core Qt6::DBus)
target_compile_options(mouse-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(universal-mouse-app src/main.cpp src/ui/window.cpp src/ui/viewer.cpp src/ui/design.cpp assets/native.qrc)
target_link_libraries(universal-mouse-app PRIVATE mouse-core Qt6::Widgets Qt6::Concurrent Qt6::Svg)
target_compile_options(universal-mouse-app PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
  add_executable(core-tests tests/core_test.cpp)
  target_link_libraries(core-tests PRIVATE mouse-core Qt6::Test)
  add_test(NAME core COMMAND core-tests)
  add_executable(window-tests tests/window_test.cpp src/ui/window.cpp src/ui/viewer.cpp src/ui/design.cpp assets/native.qrc)
  target_link_libraries(window-tests PRIVATE mouse-core Qt6::Widgets Qt6::Concurrent Qt6::Svg Qt6::Test)
  add_test(NAME window COMMAND window-tests)
  set_tests_properties(window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;UMA_ISOLATED_TEST=1")
endif()
include(GNUInstallDirs)
install(TARGETS universal-mouse-app RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/universal-mouse-app.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/universal-mouse-app.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps RENAME universal-mouse-app.svg)
install(FILES packaging/70-universal-mouse-app.rules DESTINATION lib/udev/rules.d)
install(PROGRAMS packaging/install-user.sh packaging/uninstall-user.sh DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/universal-mouse-app)
install(FILES LICENSE packaging/THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/universal-mouse-app)
set(CPACK_PACKAGE_NAME universal-mouse-app)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "MSedgeMC")
set(CPACK_GENERATOR "TGZ;DEB")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
# SVG icons load through Qt's image plugin, so shlibdeps cannot discover it.
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libqt6svg6 (>= 6.4.0)")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "qt6-wayland")
set(CPACK_DEBIAN_PACKAGE_SUGGESTS "input-remapper, imwheel")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/postinst")
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_DEBIAN_PACKAGE_DESCRIPTION "Native Qt mouse configurator with experimental Instant GM320 support")
include(CPack)
